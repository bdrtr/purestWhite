# rust/firmware's input-check image polls the buttons and the touch panel from the bare-metal
# firmware every 15 ms and counts what they give, then sleeps deeply once Power is held, as the
# C++ firmware does; Power wakes it into this firmware, which saves the report with what woke the
# chip.
# Settings > System > SD Card Firmware Update flashes it into the other OTA slot.
#   cd rust/firmware; cargo build --release; cd ../..
#   espflash save-image --chip esp32s3 --flash-size 16mb rust/firmware/target/xtensa-esp32s3-none-elf/release/input-check build/baremetal/input-check.bin
#   emu/compare.sh emu/baremetal-input.txt x4pro build/baremetal/input-check.bin
# The base side fails at its first wait for the image's output: the unmodified base firmware
# never starts the image.
ready
touch:240,562
wait-screen
touch:240,145
wait-screen
touch:240,510
wait-screen
touch:240,138
wait-quiet:3
touch:240,472
timeout:180
wait-log:input: ready
press:up
press:down
# A tap, a long press (whose lift does not tap) and a swipe.
touch:240,400
touch:240,400:900
swipe:left
press:home
press:power:1500
wait-log:sleep: deep, power
wait:3
press:power:2000
ready
expect-log:touch: 0x5d
expect-log:keys: left 1 right 1 power 1
expect-log:down 3 tap 1 swipe 1 long 1 home 1/0
# The long press's lift is suppressed: two lifts, both taken as gestures.
expect-log:lift 2 slow 0 other 0 failed 0
# The ROM's boot line shows the deep sleep's wake; its timer is 60 s, so the press woke it. The
# C++ firmware's "wake:" and "reset:" lines are not checked: the console can cut them with
# the next log line.
expect-log:rst:0x5 \(DSLEEP\)
wait-quiet:5
shot:back-home
